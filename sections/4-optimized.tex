\section{Implementação Otimizada (v3)}

\begin{frame}{v3 --- A Correção: Carregar Uma Vez, Computar Muitas Vezes}
    \textbf{Mudança-chave:} carrega todos os 5,75 milhões de pontos em \texttt{double[][]}
    \emph{antes} do loop de iteração. O CSV é lido exatamente uma vez.

    \medskip
    \begin{table}
        \centering
        \small
        \begin{tabular}{ll}
            \toprule
            \textbf{Componente} & \textbf{Mudança} \\
            \midrule
            \texttt{CsvDataset}      & novo \texttt{loadAllPoints()} $\to$ \texttt{double[][]} \\
            \texttt{KMeans}          & nova assinatura \texttt{cluster(double[][], ...)} \\
            \texttt{KMeans}          & \texttt{CountDownLatch(N)} em vez de lista de \texttt{Future} \\
            \texttt{KMeans}          & $N$ chunks fixos em vez de lotes \texttt{BATCH\_SIZE} \\
            \texttt{KMeansBenchmark} & pontos pré-carregados em \texttt{@Setup(Level.Trial)} \\
            \texttt{KMeansSampler}   & \texttt{static volatile sharedPoints} (double-checked locking) \\
            \bottomrule
        \end{tabular}
    \end{table}
\end{frame}

\begin{frame}{v3 --- Perfil JFR}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{I/O}
        \begin{itemize}
            \item \alert{0 eventos \texttt{jdk.FileRead}}
            \item CSV lido apenas na inicialização
            \item Todas as iterações: somente memória
        \end{itemize}

        \bigskip
        \textbf{GC}
        \begin{itemize}
            \item 60 eventos (vs 1.000+ na v2)
            \item 1.024 ms total de pausas
            \item Heap máximo: \alert{$\approx$2,7 GB}
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{O que domina agora?}

        \medskip
        \texttt{KMeans.squaredDistance} e \texttt{closestCentroid} dominam o
        perfil --- o custo computacional real, finalmente visível.

        \bigskip
        \textbf{Trade-off:}

        $\approx$250 MB heap (v2) $\to$ $\approx$2,7 GB (v3)
    \end{columns}
\end{frame}
