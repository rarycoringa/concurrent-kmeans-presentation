\section{Introdução}

\begin{frame}{O Algoritmo K-Means}
    \begin{itemize}
        \item Agrupa \(N\) pontos em \(k\) clusters, associando cada ponto ao centróide mais próximo
        \item Opera em duas etapas que se repetem:
            \begin{itemize}
                \item \textbf{Atribuição} — cada ponto é associado ao centróide mais próximo
                \item \textbf{Atualização} — cada centróide é recalculado como média dos seus pontos
            \end{itemize}
        \item Para quando os centróides estabilizam ou o limite de iterações é atingido
        \item A etapa de atribuição é \alert{independente entre pontos} --- candidato natural à paralelização
    \end{itemize}
\end{frame}

\begin{frame}{Dataset}
    \begin{itemize}
        \item Pedidos sintéticos gerados artificialmente
        \item \alert{$\approx$5,75 milhões} de linhas $\times$ 38 features numéricas
        \item Tamanho em disco: $\approx$1 GB
        \item Tamanho carregado em memória: $\approx$1,75 GB
    \end{itemize}

    \vspace{0.5em}
    \begin{block}{Parâmetros fixos nos experimentos}
        $k = 10$ clusters \quad $\cdot$ \quad \texttt{maxIter} = 3--20 \quad $\cdot$ \quad 38 dimensões
    \end{block}
\end{frame}

\begin{frame}{Erlang --- Características}
    \begin{itemize}
        \item Linguagem funcional, roda na máquina virtual BEAM
        \item Processos leves e isolados --- sem memória compartilhada
        \item Comunicação exclusiva por troca de mensagens
        \item GC por processo, sem pausas globais (\textit{stop-the-world})
    \end{itemize}
\end{frame}

\begin{frame}{Erlang --- Diferenças para Java}
    \begin{itemize}
        \item Sem \texttt{volatile}, mutexes ou semáforos
        \item Race conditions \alert{impossíveis por construção} --- cada processo tem sua própria memória
        \item Escala naturalmente para sistemas distribuídos sem mudar o modelo de programação
    \end{itemize}

    \vspace{1em}
    \begin{block}{Por que usar Erlang neste projeto?}
        Comparar dois modelos fundamentalmente diferentes de concorrência:
        memória compartilhada (Java) vs.\ troca de mensagens (Erlang).
    \end{block}
\end{frame}
